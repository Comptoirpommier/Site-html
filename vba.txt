Sub GenerateHTML()

                 Dim categorie As String

                ' Conteneur de l'image

               ' Définir imgFile en fonction de la présence de fichiers JPG ou PNG
               If FileExists(folderPath & ReplaceInvalidChars(productName, False) & ".jpg") Then

               ElseIf FileExists(folderPath & ReplaceInvalidChars(productName, False) & ".png") Then
                  imgFile = "images/" & ReplaceInvalidChars(productName, False) & ".png"
               Else

               End If

                    ' Convertir les espaces en %20 pour l'URL

                Print #fileNum, "<div class='line-container'>"
                Print #fileNum, "<div class='line comentarios'><strong style='color: gray;'>Comentarios: </strong>" & Replace(ws.Cells(cell.row, "O").Value, "'", "&#39;") & "</div>"
                Print #fileNum, "</div>"

                Print #fileNum, "<div class='line harmonizacao'><strong style='color: gray;'>Harmonização: </strong>" & Replace(ws.Cells(cell.row, "P").Value, "'", "&#39;") & "</div>"
                Print #fileNum, "</div>"

                Print #fileNum, "<div class='line preco'><strong style='color: gray;'>No Cartão até 3x sem Juros:</strong><strong style='color: black;'> R$ " & Replace(ws.Cells(cell.row, "Q").Value, "'", "&#39;") & "</strong></div>"
                Print #fileNum, "</div>"

    Print #fileNum, "            }"

    Print #fileNum, "    var categoria = getUrlParameter('categoria');"

' Fonction pour vérifier si un fichier existe
Function FileExists(filePath As String) As Boolean
    FileExists = (Dir(filePath) <> "")
End Function

' Fonction pour remplacer les caractères invalides dans les noms de fichiers
Function ReplaceInvalidChars(inputString As String, Optional replaceSpaces As Boolean = True) As String
    ' Remplacer certains caractères spéciaux par des tirets bas ou des entités HTML appropriées
    Dim specialChars As Variant
    specialChars = Array(":", "\", "/", "*", "?", """", "<", ">", "|", ",", ".", "%", "'", "°", "á", "é", "í", "ó", "ú", "ã", "õ", "ç")
    
    Dim replacementChars As Variant
    replacementChars = Array("_", "_", "_", "_", "_", "_", "_", "_", "_", "_", "_", "_", "_", "_", "&aacute;", "&eacute;", "&iacute;", "&oacute;", "&uacute;", "&atilde;", "&otilde;", "&ccedil;")
    
    Dim J As Integer
    For J = LBound(specialChars) To UBound(specialChars)
        inputString = Replace(inputString, specialChars(J), replacementChars(J))
    Next J

 ' Remplacer les espaces par des underscores si l'option est activée   
    If replaceSpaces Then
        inputString = Replace(inputString, " ", "_")
    End If
    
    ReplaceInvalidChars = inputString
End Function

' Fonction pour encoder une URL
Function EncodeURL(inputString As String) As String
    Dim J As Integer
    Dim char As String
    Dim encodedString As String
    
    encodedString = ""
    
    For J = 1 To Len(inputString)
        char = Mid(inputString, J, 1)
        Select Case char
            Case "%"
                encodedString = encodedString & "%25"
            Case "/"
                encodedString = encodedString & "%2F"
            Case "°"
                encodedString = encodedString & "%B0"
            Case Else
                encodedString = encodedString & char
        End Select
    Next J
    
    EncodeURL = encodedString
End Function
